// Task Detail

Context stays
with the task.

Each task can carry its description, notes, code snippet, tags, priority, deadline, and activity in one detail panel.

- Code snippets in task detail panels
- Tags, notes, and activity per task
- JetBrains Mono throughout the UI
- Built with Tailwind, vanilla JS, Firebase Auth, and Firestore
task-detail.js
// Task detail panel - code snippet
const createTask = (options) => {
  const task = {
    id: crypto.randomUUID(),
    title: options.title,
    priority: options.priority ?? 'low',
    deadline: options.deadline,
    labels: options.labels ?? [],
    completed: false,
    code: options.code ?? '',
    createdAt: new Date().toISOString()
  }
  tasks.push(task)
  renderTaskList()
  return task
}

// Initial task activity shown in the detail panel
